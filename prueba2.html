window.onload=function(){setTimeout(function(){const fechaElement=document.querySelector('#fsUpcomingTime');if(fechaElement){let fechaGMT=fechaElement.textContent.trim().replace(' (GMT)','');const fechaParts=fechaGMT.split(':');const now=new Date();now.setHours(fechaParts[0]);now.setMinutes(fechaParts[1]);now.setSeconds(0);if(!isNaN(now)){const opciones={timeZone:'America/Bogota',hour:'2-digit',minute:'2-digit',second:'2-digit'};const fechaColombia=new Intl.DateTimeFormat('es-CO',opciones).format(now);fechaElement.textContent=fechaColombia}}},2000)};
